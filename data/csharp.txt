{
    internal class Program
    {
        static void Main(string[] args)
        {
            //nested for loop
            for (int a=1; a <= 10; a++)   // a*b  //parent
            {
                for (int b = 1; b <= 10; b++) // child
                {
                    Console.Write(a*b+"\t");
                }
                Console.WriteLine();
            }

        }
    }
